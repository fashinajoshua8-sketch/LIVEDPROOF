async function requestLiveWallet(providerName) { try { let provider = null; // 1. Check for standard injected extension / in-app browser providers if (providerName === 'phantom') { provider = window.phantom?.solana || (window.solana?.isPhantom ? window.solana : null); } else if (providerName === 'solflare') { provider = window.solflare; } else if (providerName === 'glow') { provider = window.glow; } if (!provider && window.solana) { provider = window.solana; } // 2. If provider is not injected (e.g. mobile Chromium/Safari browser), handle via Mobile Deep Link if (!provider) { const isMobile = /iPhone|iPad|iPod|Android/i.test(navigator.userAgent); if (isMobile) { closeWalletModal(); const currentUrl = window.location.href; const encodedUrl = encodeURIComponent(currentUrl); const refUrl = encodeURIComponent(window.location.origin); let deepLink = ""; if (providerName === 'phantom') { deepLink = `https://phantom.app/ul/browse/${encodedUrl}?ref=${refUrl}`; } else if (providerName === 'solflare') { deepLink = `https://solflare.com/ul/v1/browse/${encodedUrl}?ref=${refUrl}`; } else if (providerName === 'glow') { deepLink = `https://glow.app/ul/browse/${encodedUrl}`; } if (deepLink) { // Triggers mobile wallet app opening with the current dApp session window.location.href = deepLink; return; } } alert(`${providerName} wallet extension or provider not detected.`); closeWalletModal(); return; } // 3. Connect via standard injected provider (PC Extensions / In-app Webviews) const res = await provider.connect(); db.currentUser.wallet = (res.publicKey || provider.publicKey).toString(); await syncProfileToSupabase(); renderLivedCard(); closeWalletModal(); } catch (err) { closeWalletModal(); console.warn("Wallet connection cancelled or failed:", err); } }
